function (tau2val, Y, vi, X, k, objective, verbose = FALSE, digits = 4) 
{
    mstyle <- .get.mstyle()
    if (any(tau2val + vi < 0)) 
        stop(mstyle$stop("Some marginal variances are negative."), 
            call. = FALSE)
    W <- diag(1/(vi + tau2val), nrow = k, ncol = k)
    stXWX <- .invcalc(X = X, W = W, k = k)
    P <- W - W %*% X %*% stXWX %*% crossprod(X, W)
    RSS <- crossprod(Y, P) %*% Y
    if (verbose) 
        cat(mstyle$verbose(paste("tau2 =", fmtx(tau2val, digits[["var"]], 
            addwidth = 4), "  RSS - objective =", fmtx(RSS - 
            objective, digits[["var"]], flag = " "), "\n")))
    return(RSS - objective)
}
